name: Compilar APK - Lotomania V5
on:
  workflow_dispatch:
  push:
    branches: [main, master]
jobs:
  construir:
    runs-on: ubuntu-latest
    steps:
      - name: Baixar projeto
        uses: actions/checkout@v4
      - name: Configurar Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - name: Configurar Android SDK
        uses: android-actions/setup-android@v3
      - name: Instalar SDK 35
        run: |
          yes | sdkmanager --licenses >/dev/null || true
          sdkmanager "platform-tools" "platforms;android-35" "build-tools;35.0.0"
      - name: Configurar Gradle 8.9
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'
      - name: Conferir estrutura
        run: |
          pwd
          find . -maxdepth 8 -type f | sort
          test -f settings.gradle
          test -f build.gradle
          test -f gradle.properties
          test -f app/build.gradle
          test -f app/src/main/AndroidManifest.xml
          test -f app/src/main/java/com/lotomania/metrica20v5/MainActivity.java
      - name: Compilar APK
        run: gradle clean :app:assembleDebug --stacktrace --no-daemon
      - name: Salvar APK
        uses: actions/upload-artifact@v4
        with:
          name: LOTOMANIA-V5-METRICA-20-APK
          path: app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
